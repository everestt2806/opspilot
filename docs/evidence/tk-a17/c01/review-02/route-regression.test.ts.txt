import { mkdtempSync, rmSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'

import { afterEach, describe, expect, it } from 'vitest'

import { buildSourceTree } from '../detectors/sourceTree'
import { resolveCollectorAppPath } from './pipeline'

const fixtures: string[] = []

function expressFixture(route: string): string {
  const root = mkdtempSync(join(tmpdir(), 'opspilot-c01-review2-'))
  fixtures.push(root)
  writeFileSync(join(root, 'package.json'), '{"dependencies":{"express":"4"}}')
  writeFileSync(join(root, 'server.js'), `${route}\napp.get('/health', handler)\n`)
  return root
}

afterEach(() => {
  while (fixtures.length > 0) rmSync(fixtures.pop()!, { recursive: true, force: true })
})

describe('C01 review 02 business probe verification', () => {
  it('falls back to health when items is write-only', () => {
    const root = expressFixture("app.post('/items', handler)")
    expect(resolveCollectorAppPath('express', buildSourceTree(root), '/health')).toBe('/health')
  })

  it('falls back to health when only a parameterized item route exists', () => {
    const root = expressFixture("app.get('/items/:id', handler)")
    expect(resolveCollectorAppPath('express', buildSourceTree(root), '/health')).toBe('/health')
  })
})

Result on `0d15eb5`: 2 tests failed. Both received `/items?limit=1` instead of `/health`.
